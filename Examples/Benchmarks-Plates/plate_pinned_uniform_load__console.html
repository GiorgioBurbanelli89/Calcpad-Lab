1% =============================================================================
2% PLACA CUADRADA EMPOTRADA EN BORDES CON CARGA UNIFORME
3% =============================================================================
4%
5% Benchmark portado de awatif-v2:
6% C:\Users\j-b-j\Documents\awatif-v2\awatif-fem\src\deform.test.ts
7% test("Plate: Rectangular pin-supported plate with uniform load")
8%
9% Este script es MATLAB 100% portable:
10% - Corre en MATLAB nativo → muestra figuras con colormap jet (SAP2000-style)
11% - Corre en Calcpad Lab → HTML con Three.js rainbow (mismo look)
12% - Corre en Octave → figuras con colormap jet
13%
14% Geometría: placa cuadrada 10×10 m, espesor 0.15 m
15% Material: E=10 GPa, ν=0.25, isotrópica
16% Carga: p₀ = -1000 N/m² (uniforme transversal)
17% Apoyos: fijos en los 4 bordes
18% Deflexión: max ≈ 12.69 mm (FEM awatif)
19% 13.541 mm (solución analítica Timoshenko exacta)
20% =============================================================================
21
22%% Parámetros del problema
23a = 10.0 % longitud en x [m]
24b = 10.0 % longitud en y [m]
25h = 0.15 % espesor [m]
26p0 = -1000 % carga distribuida [N/m²]
27E = 1.0e10 % módulo elástico [Pa]
28nu = 0.25 % coeficiente de Poisson
29
30%% Generación del mesh — grid regular 10×10 (100 nodos)
31nPts = 100;
32x = zeros(1, nPts);
33y = zeros(1, nPts);
34for k = 1:nPts
35 i = mod(k - 1, 10);
36 j = (k - i - 1) / 10;
37 x(k) = i * a / 9;
38 y(k) = j * b / 9;
39end
40
41%% Triangulación Delaunay automática (~162 triángulos)
42tri = delaunay(x, y);
43
44%% Solución analítica de Timoshenko — serie de Fourier completa
45% w(x, y) = (16·p₀ / π⁶·D) · Σ_{m,n impares} sin(mπx/a)·sin(nπy/b) / [m·n·(m²/a² + n²/b²)²]
46%
47% Sumamos m, n ∈ {1, 3, 5, 7} (4×4 = 16 términos) — converge rápido a la
48% solución exacta Timoshenko 13.541 mm.
49D = E * h^3 / (12 * (1 - nu^2))
50
51w = zeros(1, nPts);
52for k = 1:nPts
53 s = 0;
54 for m = 1:2:7
55 for n = 1:2:7
56 num = sin(m*pi*x(k)/a) * sin(n*pi*y(k)/b);
57 den = m * n * (m^2/a^2 + n^2/b^2)^2;
58 s = s + num / den;
59 end;
60 end;
61 w(k) = 16 * p0 / (pi^6 * D) * s;
62end;
63
64%% Deflexión máxima en milímetros
65w_max_mm = abs(min(w)) * 1000
66
67%% Visualización 2D — mesh wireframe
68triplot(tri, x, y);
69
70%% Visualización 3D — superficie deformada con colormap JET (estilo SAP2000)
71trisurf(tri, x, y, w);
72colormap('jet'); % azul → cian → verde → amarillo → rojo (SAP2000-style)
73colorbar(); % barra de color con valores Min/Max
74shading('interp'); % suavizar gradiente entre triángulos
75view(45, 30); % vista oblicua orbital
76axis('equal'); % escalas iguales en X, Y, Z
